cmake_minimum_required(VERSION 3.13)

if(NOT DEFINED ENV{PICO_SDK_PATH})
    message(FATAL_ERROR "PICO_SDK_PATH must point to a Pico SDK checkout")
endif()

include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(smz_input_link_test C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
pico_sdk_init()

set(LINK_TEST_ROLE "" CACHE STRING "Firmware role: MASTER or CLIENT")
set_property(CACHE LINK_TEST_ROLE PROPERTY STRINGS MASTER CLIENT)
if(NOT LINK_TEST_ROLE STREQUAL "MASTER" AND NOT LINK_TEST_ROLE STREQUAL "CLIENT")
    message(FATAL_ERROR "Set -DLINK_TEST_ROLE=MASTER or -DLINK_TEST_ROLE=CLIENT")
endif()

add_executable(smz_input_link_test link_test.c)
if(LINK_TEST_ROLE STREQUAL "MASTER")
    target_compile_definitions(smz_input_link_test PRIVATE LINK_TEST_MASTER=1)
else()
    target_compile_definitions(smz_input_link_test PRIVATE LINK_TEST_CLIENT=1)
endif()
target_compile_options(smz_input_link_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(smz_input_link_test PRIVATE pico_stdlib hardware_uart)
pico_enable_stdio_usb(smz_input_link_test 1)
pico_enable_stdio_uart(smz_input_link_test 0)
pico_add_extra_outputs(smz_input_link_test)
