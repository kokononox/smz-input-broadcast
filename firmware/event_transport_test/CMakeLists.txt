cmake_minimum_required(VERSION 3.13)

if(NOT DEFINED ENV{PICO_SDK_PATH})
    message(FATAL_ERROR "PICO_SDK_PATH must point to a Pico SDK checkout")
endif()

include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(smz_event_transport_test C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
pico_sdk_init()

# This test firmware is C-only and does not use dynamic allocation. Avoid
# pulling the SDK's C++ new/delete shim into every image, which requires the
# large ARM libstdc++ runtime package just for this C test.
if(TARGET pico_cxx_options)
    get_target_property(PICO_CXX_OPTION_SOURCES pico_cxx_options INTERFACE_SOURCES)
    if(PICO_CXX_OPTION_SOURCES)
        list(FILTER PICO_CXX_OPTION_SOURCES EXCLUDE REGEX "/new_delete\\.cpp$")
        set_property(TARGET pico_cxx_options PROPERTY INTERFACE_SOURCES "${PICO_CXX_OPTION_SOURCES}")
    endif()
endif()

set(EVENT_TEST_ROLE "" CACHE STRING "Firmware role: MASTER, CLIENT, HOST_MASTER, or HOST_MIRROR")
set_property(CACHE EVENT_TEST_ROLE PROPERTY STRINGS MASTER CLIENT HOST_MASTER HOST_MIRROR)
if(NOT EVENT_TEST_ROLE STREQUAL "MASTER" AND NOT EVENT_TEST_ROLE STREQUAL "CLIENT" AND NOT EVENT_TEST_ROLE STREQUAL "HOST_MASTER" AND NOT EVENT_TEST_ROLE STREQUAL "HOST_MIRROR")
    message(FATAL_ERROR "Set -DEVENT_TEST_ROLE=MASTER, CLIENT, HOST_MASTER, or HOST_MIRROR")
endif()

add_executable(smz_event_transport_test event_transport_test.c)
set_property(TARGET smz_event_transport_test PROPERTY LINKER_LANGUAGE C)
if(EVENT_TEST_ROLE STREQUAL "MASTER")
    target_compile_definitions(smz_event_transport_test PRIVATE EVENT_TEST_MASTER=1)
elseif(EVENT_TEST_ROLE STREQUAL "CLIENT")
    target_compile_definitions(smz_event_transport_test PRIVATE EVENT_TEST_CLIENT=1)
elseif(EVENT_TEST_ROLE STREQUAL "HOST_MASTER")
    target_compile_definitions(smz_event_transport_test PRIVATE EVENT_TEST_HOST_MASTER=1)
else()
    target_compile_definitions(smz_event_transport_test PRIVATE EVENT_TEST_HOST_MIRROR_MASTER=1)
endif()
target_compile_options(smz_event_transport_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(smz_event_transport_test PRIVATE pico_stdlib hardware_uart)
pico_enable_stdio_usb(smz_event_transport_test 1)
pico_enable_stdio_uart(smz_event_transport_test 0)
pico_add_extra_outputs(smz_event_transport_test)
